#+ Operaciones aritmeticas: suman,resta,multiplicacion y division entre variables o constantes +#

#+ Condicion simple: comparadores ==, <, >, <=, >=, != +#
if (x > y)
{
    write("x es mayor que y")
}

#+ If / else: rama alternativa que se ejecuta cuando la condicion es falsa +#
if (resultado >= 10)
{
    write("resultado alto")
}
else
{
    write("resultado bajo")
}

#+ Condicion multiple con AND: ambas condiciones simples deben cumplirse +#
if (x > 0 AND y > 0)
{
    write("ambos valores son positivos")
}

#+ Condicion multiple con OR: alcanza con que se cumpla una de las dos +#
if (x > 100 OR y > 100)
{
    write("alguno de los dos supera 100")
}

#+ Condicion con NOT: invierte el resultado de una condicion simple +#
if (NOT x > y)
{
    write("x no es mayor que y")
}

#+ While: repite el bloque mientras la condicion sea verdadera +#
while (contador < limite)
{
    write(contador)
    contador := contador + 1
}

#+ Anidamiento: if dentro de otro if, para distinguir subcasos de una misma condicion +#
if (x > 0)
{
    if (x > 10)
    {
        write("x es positivo y mayor a 10")
    }
    else
    {
        write("x es positivo pero menor o igual a 10")
    }
}

#+ Anidamiento: while dentro de un if, se ejecuta solo si la condicion inicial se cumple +#
if (contador < limite)
{
    while (contador < limite)
    {
        write("completando hasta el limite")
        contador := contador + 1
    }
}

#+ Anidamiento: if dentro de un while, evalua una condicion distinta en cada iteracion +#
contador := 0
while (contador < limite)
{
    if (contador == 0)
    {
        write("primera iteracion")
    }
    else
    {
        write("iteracion siguiente")
    }
    contador := contador + 1
}

#+ TE1 - Los tres tipos de patron en la misma estructura, con rama por defecto +#
when (x)
{
    is 10 { write("diez") }
    in 1 .. 9 { write("entre 1 y 9") }
    is < 0 { write("negativo") }
    else { write("otro valor") }
}

#+ TE1 - Patrones con constantes negativas +#
when (y)
{
    is -10 { write("menos diez") }
    in -5 .. -1 { write("entre menos 5 y menos 1") }
    else { write("fuera de los rangos") }
}

#+ TE1 - Expresion de control compuesta, sin rama por defecto +#
when (x + y * 2)
{
    is 0 { write("cero") }
    in 1 .. 100 { write("hasta 100") }
    is >= 101 { write("mas de 100") }
}

#+ TE1 - Patrones solapados: con contador = 16 gana la primera rama aunque la segunda tambien se cumple +#
contador := 16
when (contador)
{
    in 0 .. 17 { write("menor de edad") }
    is > 15 { write("mayor de 15") }
    else { write("adulto") }
}

#+ TE1 - Ninguna rama se satisface y no hay rama por defecto: no se ejecuta nada +#
when (limite)
{
    is 100 { write("cien") }
    in 200 .. 300 { write("entre 200 y 300") }
}

#+ TE1 - Control Float con patrones numericos, y control String con valor exacto +#

#+ TE1 - when anidado dentro de un while, con un if dentro de una rama +#
contador := 0
while (contador < limite)
{
    when (contador * 2)
    {
        in 0 .. 5 { write("bajo") }
        is >= 6
        {
            if (contador == 4)
            {
                write("caso especial")
            }
            else
            {
                write("alto")
            }
        }
        else { write("otro") }
    }
    contador := contador + 1
}

#+ TE1 - when dentro de otro when: cada uno usa su propia variable auxiliar +#
when (x)
{
    is 1
    {
        when (y)
        {
            is 2 { write("x es 1 e y es 2") }
            else { write("x es 1 e y no es 2") }
        }
    }
    else { write("x no es 1") }
}

#+ TE3 - Potencia simple con base y exponente constantes: 2 ** 3 = 8 +#
p1 := 2 ** 3

#+ TE3 - Potencia con variables: base ** tope = 2 ** 3 = 8 +#
base := 2
tope := 3
p2 := base ** tope

#+ TE3 - Potencia encadenada sin parentesis, asocia a derecha: 2 ** (3 ** 2) = 512 +#
p3 := 2 ** 3 ** 2

#+ TE3 - Los parentesis alteran el orden: (2 ** 3) ** 2 = 64 +#
p4 := (2 ** 3) ** 2

#+ TE3 - Precedencia: la potencia se resuelve antes que * y +, 2 + 3 * 2 ** 2 = 14 +#
p5 := 2 + 3 * 2 ** 2

#+ TE3 - El menos unario tiene menor precedencia que la potencia: -2 ** 2 = -4 +#
p6 := -2 ** 2

#+ TE3 - Exponente cero: 5 ** 0 = 1, cualquiera sea la base +#
p7 := 5 ** 0

#+ TE3 - El tipo del resultado es el de la base: base Float da resultado Float, 1.5 ** 2 = 2.25 +#
resultado := 1.5 ** 2

#+ TE3 - Comparacion de tres vias con sus tres resultados: -1, 1 y 0 +#
m := 3
n := 7
r := m <=> n
r := n <=> m
r := m <=> 3

#+ TE3 - Operandos con expresiones aritmeticas: compara 4 contra 4, r = 0 +#
r := m + 1 <=> n - 3

#+ TE3 - El resultado es un entero reutilizable en cualquier expresion: (m <=> n) * 10 = -10 +#
r := (m <=> n) * 10

#+ TE3 - El resultado de <=> dentro de la condicion de un if: (m <=> n) == 0 equivale a m == n +#
if ((m <=> n) == 0)
{
    write("m y n son iguales")
}
else
{
    write("m y n son distintos")
}

#+ TE3 - La potencia dentro de la condicion de un while, se recalcula en cada vuelta +#
base := 1
tope := 20
while (base ** 2 < tope)
{
    write(base)
    base := base + 1
}
